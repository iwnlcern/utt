```python
"""XXH3-128 (seed 0) via the reference xxhsum CLI; canonical big-endian bytes (DD §4.5)."""
import shutil
import subprocess


def xxh3_128(data: bytes) -> bytes:
    if shutil.which("xxhsum") is None:
        raise RuntimeError("xxhsum CLI not found (install xxHash >= 0.8)")
    out = subprocess.run(["xxhsum", "-H2", "-"], input=data, capture_output=True, check=True)
    hexdigest = out.stdout.split()[0].decode()
    if len(hexdigest) != 32:
        raise RuntimeError(f"unexpected xxhsum output: {out.stdout!r}")
    return bytes.fromhex(hexdigest)
```

```python
# theory/certschema/game.py
"""Alternating rule-lock game (DD §2, Appendix A): state identity, D4 canonical form."""
from dataclasses import dataclass

D4 = {
    "e":    (0, 1, 2, 3, 4, 5, 6, 7, 8),
    "r90":  (2, 5, 8, 1, 4, 7, 0, 3, 6),
    "r180": (8, 7, 6, 5, 4, 3, 2, 1, 0),
    "r270": (6, 3, 0, 7, 4, 1, 8, 5, 2),
    "mh":   (2, 1, 0, 5, 4, 3, 8, 7, 6),
    "mv":   (6, 7, 8, 3, 4, 5, 0, 1, 2),
    "md":   (0, 3, 6, 1, 4, 7, 2, 5, 8),
    "ma":   (8, 5, 2, 7, 4, 1, 6, 3, 0),
}
LINES = (0b000000111, 0b000111000, 0b111000000,
         0b001001001, 0b010010010, 0b100100100,
         0b100010001, 0b001010100)
ANY = 9


def _permute_mask(mask: int, g: tuple) -> int:
    out = 0
    for c in range(9):
        if mask >> c & 1:
            out |= 1 << g[c]
    return out


@dataclass(frozen=True)
class State:
    x: tuple
    o: tuple
    forced: int

    def serialize(self) -> bytes:
        b = b"".join(m.to_bytes(2, "little") for m in self.x)
        b += b"".join(m.to_bytes(2, "little") for m in self.o)
        return b + bytes([self.forced])

    def counts(self) -> tuple:
        return (sum(m.bit_count() for m in self.x), sum(m.bit_count() for m in self.o))

    def side_to_move(self) -> str:
        cx, co = self.counts()
        return "X" if cx == co else "O"

    def board_closed(self, b: int) -> bool:
        return _won(self.x[b]) or _won(self.o[b]) or ((self.x[b] | self.o[b]).bit_count() == 9)

    def well_formed(self) -> bool:
        for b in range(9):
            if self.x[b] >> 9 or self.o[b] >> 9 or (self.x[b] & self.o[b]):
                return False
        cx, co = self.counts()
        if not (0 <= cx - co <= 1):
            return False
        if not (0 <= self.forced <= 9):
            return False
        if self.forced != ANY and self.board_closed(self.forced):
            return False
        return True


def _won(mask: int) -> bool:
    return any(mask & line == line for line in LINES)


def transform(s: State, g: tuple) -> State:
    nx, no = [0] * 9, [0] * 9
    for b in range(9):
        nx[g[b]] = _permute_mask(s.x[b], g)
        no[g[b]] = _permute_mask(s.o[b], g)
    nf = ANY if s.forced == ANY else g[s.forced]
    return State(x=tuple(nx), o=tuple(no), forced=nf)


def canonicalize(s: State) -> tuple:
    best, best_g = None, None
    for g in D4.values():  # fixed order; first minimum wins ties (deterministic)
        t = transform(s, g)
        ser = t.serialize()
        if best is None or ser < best.serialize():
            best, best_g = t, g
    return best, best_g
```

```python
def legal_moves(s: State):
    if terminal(s) is not None:
        return []
    boards = [s.forced] if s.forced != ANY else [b for b in range(9) if not s.board_closed(b)]
    moves = []
    for b in boards:
        occ = s.x[b] | s.o[b]
        for c in range(9):
            if not occ >> c & 1:
                moves.append((b, c))
    return moves


def apply_for(s: State, b: int, c: int, side: str) -> State:
    # Explicit-mover apply: fixture states need not respect alternation parity.
    if side not in ("X", "O"):
        raise ValueError(side)
    if (b, c) not in legal_moves(s):
        raise ValueError(f"illegal move ({b},{c})")
    x, o = list(s.x), list(s.o)
    (x if side == "X" else o)[b] |= 1 << c
    child = State(x=tuple(x), o=tuple(o), forced=c)
    if child.board_closed(c):
        child = State(x=child.x, o=child.o, forced=ANY)
    return child


def apply_move(s: State, b: int, c: int) -> State:
    return apply_for(s, b, c, s.side_to_move())


def board_result(s: State, b: int):
    # v1 closure-record vocabulary: "X" | "O" | "full" | None (open)
    if _won(s.x[b]):
        return "X"
    if _won(s.o[b]):
        return "O"
    if (s.x[b] | s.o[b]).bit_count() == 9:
        return "full"
    return None


def _macro_win(s: State, side: str) -> bool:
    masks = s.x if side == "X" else s.o
    macro = 0
    for b in range(9):
        if _won(masks[b]):
            macro |= 1 << b
    return _won(macro)


def terminal(s: State):
    if _macro_win(s, "X"):
        return "X"
    if _macro_win(s, "O"):
        return "O"
    if all(s.board_closed(b) for b in range(9)):
        return "draw"
    return None
```

CAUTION (§Appendix A rule 5 ordering): macro win is checked before all-closed, matching `closure-double-line-macro-precedes-all-closed` in the fixtures.
CAUTION (`apply_move`): closure must be evaluated on the CHILD (the move may close board c itself → ANY), which the code above does.

```python
# theory/certschema/wire.py
"""DD §4/§7 byte builders. All integers LE; digests = canonical XXH3-128 (xxh.py)."""
from dataclasses import dataclass

from .game import State
from .xxh import xxh3_128

MAGIC_CERT = b"UTTTCERT"
MAGIC_VERDICT = b"UTTTVRDT"
SCHEMA = (0, 1)
RULESET_ID = "uttt-s2-r1"
RULESET_DIGEST = bytes.fromhex("16e85b36332493470fa1006516128579")
SYMMETRY_ID = "d4-lexmin-v1"
CLAIM = {"WIN_X": 0, "WIN_O": 1, "NOLOSS_X": 2, "NOLOSS_O": 3}
KIND = {"PROVER": 0, "OPPONENT": 1, "TERMINAL": 2, "RULE": 3}


def _u16(v): return v.to_bytes(2, "little")
def _u32(v): return v.to_bytes(4, "little")
def _u64(v): return v.to_bytes(8, "little")
def _str16(s: str) -> bytes:
    e = s.encode()
    return _u16(len(e)) + e


@dataclass(frozen=True)
class Record:
    state: State
    kind: int
    payload: int  # PROVER: move 0..80 in the record state's canonical frame; else 0xFF

    def encode(self) -> bytes:
        return self.state.serialize() + bytes([self.kind, self.payload, 0x00])


def build_certificate(claim: int, root: State, plies: dict) -> bytes:
    hdr = MAGIC_CERT + _u16(SCHEMA[0]) + _u16(SCHEMA[1]) + bytes([claim, 0, 0, 0])
    hdr += _str16(RULESET_ID) + RULESET_DIGEST + _str16(SYMMETRY_ID) + root.serialize()
    hdr += _u16(0)  # rule_count: v0 empty registry
    totals = [0, 0, 0, 0]
    chunks = []
    for ply in sorted(plies, reverse=True):
        recs = plies[ply]
        for r in recs:
            totals[r.kind] += 1
        chunks.append((ply, _u32(len(recs)) + _u32(0) + b"".join(r.encode() for r in recs)))
    hdr += b"".join(_u64(t) for t in totals) + _u32(len(chunks))
    header_len = len(hdr) + 40 * len(chunks) + 16
    rows = b""
    offset = header_len
    for ply, chunk in chunks:
        rows += bytes([ply, 0]) + _u16(0) + _u32((len(chunk) - 8) // 40) + _u64(offset) + _u64(len(chunk)) + xxh3_128(chunk)
        offset += len(chunk)
    return hdr + rows + xxh3_128(rows) + b"".join(c for _, c in chunks)


def header_region(cert: bytes) -> bytes:
    # header region = everything before the first chunk = up to end of manifest_digest.
    # Recover its length from the first manifest row's byte_offset.
    # Row area starts after the variable header; find it by parsing minimally:
    i = len(MAGIC_CERT) + 4 + 4  # magic, schema, claim+reserved
    rid_len = int.from_bytes(cert[i:i + 2], "little"); i += 2 + rid_len + 16
    sid_len = int.from_bytes(cert[i:i + 2], "little"); i += 2 + sid_len
    i += 37  # root
    rc = int.from_bytes(cert[i:i + 2], "little"); i += 2
    for _ in range(rc):
        il = int.from_bytes(cert[i:i + 2], "little"); i += 2 + il + 2
    i += 32  # totals
    rows = int.from_bytes(cert[i:i + 4], "little"); i += 4
    first_offset = int.from_bytes(cert[i + 8:i + 16], "little")
    assert first_offset == i + 40 * rows + 16
    return cert[:first_offset]


def header_digest(cert: bytes) -> bytes:
    return xxh3_128(header_region(cert))


def build_verdict(game_value: int, root: State, members: list) -> bytes:
    v = MAGIC_VERDICT + _u16(SCHEMA[0]) + _u16(SCHEMA[1]) + bytes([game_value, 0, 0, 0])
    v += _str16(RULESET_ID) + RULESET_DIGEST + _str16(SYMMETRY_ID) + root.serialize()
    v += bytes([len(members)])
    for claim, path, hdigest in members:
        v += bytes([claim]) + _str16(path) + hdigest
    return v
```

NOTE: `build_certificate` emits one chunk per ply (sufficient for this corpus; multi-chunk plies stay legal per §4.1 but are not needed here).

```python
# theory/certschema/vectors.py
"""Positive vectors and (Task 6) mutants for the DD §9 conformance corpus."""
from .game import ANY, State, apply_move, canonicalize, legal_moves, terminal
from .wire import CLAIM, KIND, Record, build_certificate, build_verdict, header_digest


def _mk(xd: dict, od: dict, forced: int) -> State:
    x = [0] * 9; o = [0] * 9
    for b, cells in xd.items():
        for c in cells:
            x[b] |= 1 << c
    for b, cells in od.items():
        for c in cells:
            o[b] |= 1 << c
    return State(x=tuple(x), o=tuple(o), forced=forced)


def golden_root() -> State:
    raw = _mk({0: (0, 1, 2), 1: (0, 1, 2), 2: (0, 1, 2)},
              {3: (3, 5), 4: (1, 7), 5: (0, 8), 6: (4,), 7: (4,)}, ANY)
    return canonicalize(raw)[0]


def _prover_record(raw_parent: State, move: tuple) -> Record:
    canon_parent, g = canonicalize(raw_parent)
    b, c = move
    stored = g[b] * 9 + g[c]  # move transported into the canonical frame (DD §2.5)
    return Record(state=canon_parent, kind=KIND["PROVER"], payload=stored)


def build_p1() -> bytes:
    root = golden_root()
    return build_certificate(CLAIM["WIN_X"], root, {17: [Record(root, KIND["TERMINAL"], 0xFF)]})


def build_golden_verdict() -> bytes:
    root = golden_root()
    return build_verdict(0, root, [(CLAIM["WIN_X"], "golden-terminal-winx.utc", header_digest(build_p1()))])


def p2_states():
    g = golden_root()
    x = list(g.x); x[8] &= ~(1 << 8)
    parent = State(x=tuple(x), o=g.o, forced=8)
    return parent, (8, 8)


def build_p2() -> bytes:
    parent_raw, move = p2_states()
    child = canonicalize(apply_move(parent_raw, *move))[0]
    prover = _prover_record(parent_raw, move)
    assert terminal(child) == "X" and child == golden_root()
    return build_certificate(CLAIM["WIN_X"], prover.state,
                             {16: [prover], 17: [Record(child, KIND["TERMINAL"], 0xFF)]})


def p3_states():
    s0 = _mk({4: (4,), 5: (0, 1, 5, 6), 6: (6, 7, 8), 7: (6, 7, 8), 8: (6, 7)},
             {0: (0, 1), 1: (3, 4), 2: (5, 6), 3: (7, 8), 5: (2, 3, 4, 7)}, 5)
    return s0, (5, 8), (8, 8)


def build_p3() -> bytes:
    s0, o_move, x_move = p3_states()
    assert legal_moves(s0) == [o_move] and terminal(s0) is None
    s1 = apply_move(s0, *o_move)
    s2 = apply_move(s1, *x_move)
    assert terminal(s2) == "X"
    opp = Record(canonicalize(s0)[0], KIND["OPPONENT"], 0xFF)
    prover = _prover_record(s1, x_move)
    term = Record(canonicalize(s2)[0], KIND["TERMINAL"], 0xFF)
    return build_certificate(CLAIM["WIN_X"], opp.state, {25: [opp], 26: [prover], 27: [term]})
```

```python
# theory/certschema/generate.py
"""Deterministic corpus generator. Usage: python -m certschema.generate [--out DIR]."""
import argparse
import json
import pathlib

from .vectors import (MUTANTS, build_golden_verdict, build_mutant, build_p1,
                      build_p2, build_p3, catalogue)


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", default=str(pathlib.Path(__file__).parent))
    out = pathlib.Path(ap.parse_args().out)
    vec = out / "vectors"; mut = out / "mutants"
    vec.mkdir(parents=True, exist_ok=True); mut.mkdir(parents=True, exist_ok=True)
    (vec / "golden-terminal-winx.utc").write_bytes(build_p1())
    (vec / "golden-winx.utv").write_bytes(build_golden_verdict())
    (vec / "p2-prover-winx.utc").write_bytes(build_p2())
    (vec / "p3-opponent-winx.utc").write_bytes(build_p3())
    for mid in MUTANTS:
        (mut / f"{mid}.bin").write_bytes(build_mutant(mid))
    # Verdict-mutant member resolution (MR-2): the .utv member path resolves
    # relative to the manifest's own directory, so mutant verdicts need the
    # valid golden certificate ADJACENT to them under mutants/.
    (mut / "golden-terminal-winx.utc").write_bytes(build_p1())
    (out / "catalogue.json").write_text(json.dumps(catalogue(), indent=2, sort_keys=True) + "\n")


if __name__ == "__main__":
    main()
```
