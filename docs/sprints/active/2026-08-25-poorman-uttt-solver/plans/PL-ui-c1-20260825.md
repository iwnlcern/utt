Interface contracts used across tasks (exact names; defined in Task 3/6/8):

```ts
// log/types.ts (Task 3)
export type Mark = 'X' | 'O';
export type Cell = readonly [number, number];
export type FaultClass = 'timeout'|'eof_or_crash'|'invalid_utf8'|'invalid_json'|'schema_violation'|'wrong_request_id'|'extra_protocol_line'|'oversize_line'|'illegal_bid'|'illegal_move';
export interface RawCapture { b64: string; truncated: boolean; bytes_total: number }
export interface TurnRecord { validation: 'ok'|FaultClass; bid?: number; move?: Cell; info?: unknown; elapsed_ms: number; raw?: RawCapture }
export interface HelloRecord { validation: 'ok'|FaultClass; name?: string; version?: string; elapsed_ms: number; raw?: RawCapture }
export interface Attempt { request_id: string; attempt: number; turns: Partial<Record<Mark, TurnRecord>> }  // per-seat container field name MUST be confirmed against the approved schema at the Task 3 gate (evidence points to `turns`); every field name in this block is illustrative until that gate passes
export interface Resolution { winner: Mark; reason: 'higher_bid'|'tie_last_mover'|'tie_coin'|'fault'; coin?: Mark; payment: number; move: Cell; closures: {local: number; result: Mark|'full'}[]; macro_line?: number[]; forced_next: number|null }
export type LogEvent = GameStartEvent | AuctionEvent | RecoveryEvent | GameEndEvent;  // shapes per DD-harness-c1 §8
export interface GameRecord { start: GameStartEvent; events: LogEvent[]; end?: GameEndEvent; truncated: boolean }
export class LogError extends Error { constructor(public line: number, public eventIndex: number|null, public reason: string) { super(reason) } }

// replay/model.ts (Task 6). RecoveryEvent REQUIRES ply + trigger_request_id (harness amendment 1):
// interface RecoveryEvent { event:'recovery'; seat: Mark; ply: number; trigger_request_id: string; fault: FaultClass; hello: HelloRecord }
// Ordinal convention (MR3): wire `ply` is ZERO-based; AuctionStep_j ordinals are ONE-based (j = index+1);
// cursor k = "after k resolved auctions"; derive validates plies are sequential zero-based.
export interface Position { board: readonly string[]; budgets: Readonly<Record<Mark, number>>; forced: number|null; tieOwner: Mark|null; closed: Readonly<Record<number, Mark|'full'>> }
export interface SetupStep { start: GameStartEvent }
export interface AuctionStep { ply: number /* zero-based wire value */; pre: Position; attempts: Attempt[]; recoveries: RecoveryEvent[] /* key-attached, raw order kept */; outcome: 'resolved'|'voided'|'aborted_recovery_fault'; resolution?: Resolution; post?: Position }
export interface ReplayModel { setup: SetupStep; auctions: AuctionStep[]; trailingRecoveries: RecoveryEvent[] /* ONLY recoveries whose ply has no auction in the prefix */; terminal?: GameEndEvent; positions: Position[]; truncated: boolean }
export function deriveReplayModel(g: GameRecord): ReplayModel

// format/money.ts (Task 5)
export type Share = { kind: 'ok'; value: number } | { kind: 'na'; why: 'both budgets exhausted' }
export function share(units: number, combined: number): Share
export function formatPercent(s: Share): string           // "63.41%" | "n/a — both budgets exhausted"
export function formatUnits(units: number): string        // "634 100 000" (narrow NBSP grouping)

// analysis/extract.ts (Task 8)
export type Quality = 'exact'|'bound'|'estimate';
export type AnalysisEntry =
  | { kind: 'ok'; t?: number; criticalBid?: number; pvIfWin?: Cell; pvIfLose?: Cell; quality?: Quality; lo?: number; hi?: number; depth?: number; complete?: boolean; degraded: string[] }
  | { kind: 'unavailable'; why: string }
export const PV_PIN: { pinned: boolean; source: string | null }   // starts {pinned:false, source:null}
export function extractAnalysis(step: AuctionStep): Partial<Record<Mark, AnalysisEntry>>
```
