```python
@dataclass
class KnifePoint:
    board: str; h: str; bx: int; bo: int
    T: Fraction                     # continuous threshold at (board, h)
    discrete: str                   # "X" | "O" | "draw" from the masked discrete oracle
    convention: str                 # prediction of the ratified large-scale convention: winner == h
    agrees: bool                    # discrete == ("X" if h == "X" else "O")
    state_class: str                # "a<b" | "a=b", joined with E(s), e.g. "a=b/E3"
def sweep(N: int) -> list[KnifePoint]
# enumerate every MASKED nonterminal point with bx*q == p_T*M exactly (T = p_T/q, M = bx+bo > 0)
# at starting scale N, using the landed reachable_masks/solve outputs; deterministic order.
def spot_sweep(N: int, targets: list[tuple[str, str]], fold: Callable[[KnifePoint], None]) -> None
# M3 STREAMING DISCIPLINE: spot_sweep never materializes the point population. It solves ONE
# distinct board at a time, invokes `fold` per qualifying KnifePoint, updates nothing else,
# and releases that board's value/mask arrays before solving the next (modeled on the landed
# targeted_report board-release discipline). `report` consumes it via aggregate accumulators
# + the optional stream digest; it never concatenates KnifePoint lists. The small-scale
# `sweep(N)` helper MAY stay list-returning for focused unit tests only.
# M1 population contract (binds to the LANDED selector; no free choice):
# targets = sorted (board, h) keys from select_spot_targets(solve_continuous(), masks_32, 32)
# computed ONCE from the N=32 exhaustive masks — the landed canonical cardinality is 20,332
# targets and is preserved as artifact/acceptance evidence. Per spot scale N in {64, 128}:
# call solve_targeted(board, N) exactly once per DISTINCT board; consume ONLY the selected
# owner keys (never widen to the unselected owner); enumerate EVERY mask-true, nonterminal,
# positive-total point satisfying bx*q == p_T*(bx+bo). Hard-fail (raise) on an empty or
# missing selected mask or any population/count-conservation failure — never drop or sample.
def report(scales: list[int], spot_scales: list[int]) -> dict
# M2 deterministic JSON contract (the CLI serializes exactly this, byte-deterministically):
# {
#   "methodology": {"version": 1, "convention": "P1b tie-owner-wins", "predicate": "bx*q == p_T*M",
#                    "exhaustive_scales": [...], "spot_scales": [...], "spot_target_count": 20332,
#                    "no_sampling": true},
#   "per_scale": [{"scale": N, "mode": "exhaustive" | "targeted", "target_count": <int|null>,
#                  "equality_points": <int>, "agreements": <int>, "disagreements": <int>}],
#                  # invariant per record: equality_points == agreements + disagreements
#   "groups": [{"scale": N, "mode": ..., "h": ..., "state_class": ..., "discrete": ...,
#               "agrees": ..., "count": <int>}],   # deterministic key order; draws and
#                                                   # opponent-wins are DISTINCT rows
#   "disagreement_groups": [ ... ],  # M3: the FULL disagreement table is the complete
#                     # deterministic NONZERO AGGREGATE PARTITION, not a coordinate dump:
#                     # every group row with agrees == false (keyed scale/mode/h/state_class/
#                     # discrete, carrying count), emitted without truncation. A measured
#                     # coordinate dump would be ~157 MiB (1,194,488 rows x 138 B) and exceed
#                     # GitHub's 100 MiB object limit — forbidden, as are compression of the
#                     # same monolith, undeclared file splits, Git LFS, or external storage.
#   "stream_digests": [{"scale": N, "mode": ..., "sha256": <hex>}]
#                     # optional reproducibility anchor: per scale, sha256 over the
#                     # deterministic stream of disagreement coordinate tuples
#                     # (board, h, bx, bo) in sweep order — a digest, never sample rows.
# }
# Conservation invariants (asserted in code and tests): sum(groups.count where scale) ==
# per_scale.equality_points; sum(disagreement_groups.count where scale) ==
# per_scale.disagreements; disagreement_groups == [g for g in groups if not g.agrees, count > 0].
```
